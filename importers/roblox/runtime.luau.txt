local byTable = {}
local byRecord = {}
for _, tableData in ipairs(data.tables) do
    if tableData.uuid then byTable[tableData.uuid] = tableData end
    local rows = {}
    if tableData.uuid then byRecord[tableData.uuid] = rows end
    for _, record in ipairs(tableData.records or {}) do rows[record.id] = record end
end
local function bezier(p, t)
    local u = 1 - t
    return u*u*u*p[1] + 3*u*u*t*p[2] + 3*u*t*t*p[3] + t*t*t*p[4]
end
local function sample(self, x)
    if x ~= x then return 0 end
    local segments = self.segments
    if x <= 0 then return segments[1].y[1] end
    if x >= 1 then return self.endValue end
    local loIndex, hiIndex = 1, #segments + 1
    while loIndex < hiIndex do
        local mid = math.floor((loIndex + hiIndex) / 2)
        if segments[mid].x[4] <= x then loIndex = mid + 1 else hiIndex = mid end
    end
    local segment = segments[loIndex]
    if segment.constant or x == segment.x[1] then return segment.y[1] end
    local lo, hi = 0, 1
    for _ = 1, 40 do
        local t = (lo + hi) * 0.5
        local bx = bezier(segment.x, t)
        if bx == x then return bezier(segment.y, t) end
        if bx < x then lo = t else hi = t end
    end
    return bezier(segment.y, (lo + hi) * 0.5)
end
local function resolve(self)
    if self.id == "" then return nil end
    if self.__devcel_kind == "table_ref" then return byTable[self.id] end
    local rows = self.targetTableId and byRecord[self.targetTableId]
    return rows and rows[self.id] or nil
end
for _, tableData in ipairs(data.tables) do
    for _, record in ipairs(tableData.records or {}) do
        for _, value in pairs(record.values or {}) do
            if type(value) == "table" then
                if value.__devcel_kind == "curve" then value.Sample = sample
                elseif value.__devcel_kind == "reference" or value.__devcel_kind == "table_ref" then value.Resolve = resolve end
            end
        end
    end
end
